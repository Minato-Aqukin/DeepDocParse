func TestIndependentSnapshotPermissionEpochDoesNotRevive(t *testing.T) {
 s:=&Store{pool:testPool(t)}; ctx:=context.Background(); org:=seedOrg(t,s)
 r:=testNode("changing-visibility",true); registerApproved(t,s,org,r)
 snap,e:=s.CreateMemberSnapshot(ctx,org,"alice","alice-scope","local-node",false,1,time.Minute); if e!=nil{t.Fatal(e)}
 r.Descriptor.Revision=2; r.VisibleToOrg=false; if _,e=s.RegisterNode(ctx,org,r);e!=nil{t.Fatal(e)}
 if _,e=s.SetNodeState(ctx,org,r.Descriptor.NodeID,"approved");e!=nil{t.Fatal(e)}
 p,e:=s.MemberSnapshotPage(ctx,org,"alice","alice-scope",snap.ID,"",false);if e!=nil{t.Fatal(e)}
 if p.Members[0].State!="revoked"{t.Fatal("fixture did not revoke visibility")}
 r.Descriptor.Revision=3;r.VisibleToOrg=true;if _,e=s.RegisterNode(ctx,org,r);e!=nil{t.Fatal(e)}
 if _,e=s.SetNodeState(ctx,org,r.Descriptor.NodeID,"approved");e!=nil{t.Fatal(e)}
 p,e=s.MemberSnapshotPage(ctx,org,"alice","alice-scope",snap.ID,"",false);if e!=nil{t.Fatal(e)}
 if p.Members[0].State!="revoked"{t.Errorf("counterexample: old revoked snapshot revived state=%s descriptor_revision=%d",p.Members[0].State,p.Members[0].Descriptor.Revision)}
}
